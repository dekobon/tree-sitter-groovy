==================
A line break after yield makes it a name, as Groovy takes the value only on its line
==================

def r = switch (v) {
  case 1 -> {
    yield
    x
  }
  case 2 -> {
    yield // c
    y
  }
  case 3 -> {
    yield \
      z
  }
  default -> {
    yield /* c */ w
  }
}
